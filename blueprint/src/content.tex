\begin{lemma}[Hilbert-space and projective preliminaries]\label{prelim:flow}
  \lean{GaussianTesting.HilbertSquare, GaussianTesting.hilbertTensorSquare, GaussianTesting.dense_span_hilbertTmul, GaussianTesting.dense_span_hilbertTriple, GaussianTesting.hilbertCoords, GaussianTesting.pureFidelity, GaussianTesting.rayTangentMetric, GaussianTesting.tendsto_dist_div_time_rayOfTangent, GaussianTesting.tendsto_rayTangentMetric_of_dist, GaussianTesting.pureFidelity_sq_le_iff, GaussianTesting.rayChartHomeomorph, GaussianTesting.contDiffOn_rayTransition, GaussianTesting.rayDist_mk, GaussianTesting.rayMetricSpace, GaussianTesting.rayCompleteSpace, GaussianTesting.projective_angle_le_of_chord_bound, GaussianTesting.overlap_le_pureFidelity, GaussianTesting.pureFidelity_nonneg, GaussianTesting.pureFidelity_le_one, GaussianTesting.sq_overlap_le_pureFidelity_sq, GaussianTesting.pureFidelity_phase}
  \leanok
  \noderef{projective-angle}{Section~3, preliminaries}\par
  This node collects only the general background used by the section's proofs:
  completed Hilbert tensor products, projective geometry and path-length
  comparison, and elementary bounds for supremum fidelity.

  \textbf{Hilbert tensor products.}
  The completed tensor square and cube have dense spans of elementary tensors.
  A Hilbert basis of arbitrary cardinality identifies these spaces with
  product-indexed $\ell^2$. These constructions underlie
  \paperref{acceptance}{Eq.~(3.1)} and \paperref{three-copy}{Eq.~(3.6)};
  the test-specific pair operators and their moments belong to those result nodes.

  \textbf{Projective geometry.}
  On the open set of rays $[x]$ with $\langle u,x\rangle\ne0$, the affine
  chart sends $[x]$ to $x/\langle u,x\rangle$. These charts cover the
  quotient of nonzero vectors by nonzero scalars and have smooth changes of
  coordinates. For unit representatives $x,y$, the distance is
  $d_{\rm FS}([x],[y])=\arccos|\langle x,y\rangle|$. It induces the quotient
  topology and is complete when the Hilbert space is complete.
  At a unit representative $\psi$, the tangent metric is
  $g_\psi(v,w)=\operatorname{Re}\langle v,w\rangle$ on $\psi^\perp$.
  The Fubini--Study speed of a curve with horizontal velocity $v$ is $\norm{v}$.

  \textbf{General path-length comparison.}
  Let $\gamma$ be a continuous path of unit vectors on $[0,t]$ and $v$ a
  real-valued function. Suppose every subinterval $[s,u]\subseteq[0,t]$
  satisfies $\norm{\gamma(u)-\gamma(s)}\le v(s)-v(u)$. Then the
  \paperref{projective-angle}{projective angle} between its endpoints satisfies
  \[
    \arccos|\langle\gamma(0),\gamma(t)\rangle|\le v(0)-v(t).
  \]
  For each $k>1$, a sufficiently fine partition makes the angle between
  successive unit vectors on the underlying real unit sphere at most $k$
  times their chord length. The triangle inequality bounds the endpoint
  angle by their sum, and the decreases in $v$ telescope. Let $k$ decrease
  to one. The projective angle is at most the real unit-sphere angle.

  \textbf{Supremum fidelity.}
  For a nonempty class $\mathcal G$ of unit vectors, write
  $F_{\mathcal G}(\psi)=\sup_{\phi\in\mathcal G}|\langle\psi,\phi\rangle|$,
  as in \paperref{fidelity}{Eq.~(3.2)}. If $\psi$ is a unit vector, then
  $0\le F_{\mathcal G}(\psi)\le1$. For each
  $\phi\in\mathcal G$,
  \[
    |\langle\psi,\phi\rangle|\le F_{\mathcal G}(\psi),
    \qquad \fid{\psi}{\phi}\le F_{\mathcal G}(\psi)^2.
  \]
  Cauchy--Schwarz bounds the overlaps above, and each member of the set
  defining the supremum is bounded by that supremum. Both sides are
  nonnegative, so the comparison persists after squaring. Multiplying $\psi$
  by a scalar of unit norm preserves all absolute overlaps and their supremum.
  For a nonempty class of unit vectors, bounding the squared supremum by $r$
  is equivalent to bounding every squared overlap by $r$.
\end{lemma}
\leanproof{prelim:flow}

\begin{definition}[Acceptance projection]\label{def:acceptance}

  Let $P$ be a self-adjoint idempotent on the completed tensor square that commutes with

  $p(\psi)=\operatorname{Re}\langle\psi^{\otimes2},P\psi^{\otimes2}\rangle$.
\end{definition}

\begin{definition}[Certainty set and fidelity]\label{def:fidelity}
  \lean{GaussianTesting.certaintyStates, GaussianTesting.pureFidelity}
  \leanok
  \uses{def:acceptance}
  \noderef{fidelity}{Eq.~(3.2)}\par
  For a two-copy test with acceptance probability $p$, its certainty set is
  $\mathcal G=\{\phi:\norm{\phi}=1,\ p(\phi)=1\}$. More generally, for a
  nonempty class $\mathcal G$ of unit vectors, its fidelity with $\psi$ is
  \[
    F_{\mathcal G}(\psi)=\sup_{\phi\in\mathcal G}
      |\langle\psi,\phi\rangle|.
  \]
\end{definition}

\begin{lemma}[Eq. (3.4), Fubini--Study gradient of acceptance]\label{lem:acceptance-gradient}
  \lean{GaussianTesting.fubiniStudyGradient_rayAcceptance, GaussianTesting.TensorSquare.hasGradientAt_rayAcceptance, GaussianTesting.TensorSquare.isFubiniStudyGradient_iff, GaussianTesting.TensorSquare.gradient_rayAcceptance_phase, GaussianTesting.TensorSquare.hasFDerivAt_accProbA_rayChartSection}
  \leanok
  \uses{def:acceptance, prelim:flow}
  \noderef{gradient}{Eq.~(3.4)}\par
  At a ray $[\psi]$ with unit representative $\psi$, write the centered
  chart as $v\mapsto[\psi+v]$ for $v\in\psi^\perp$. Its tangent metric
  is $g_\psi(v,w)=\operatorname{Re}\langle v,w\rangle$. For every such $v$,
  the differential of acceptance satisfies
  \[
    dp_{[\psi]}(v)=g_\psi(G,v),\qquad
    G=4\big((I\otimes\langle\psi|)P\psi^{\otimes 2}-p([\psi])\psi\big).
  \]
  Thus $G$ is the unique horizontal vector representing the differential.
  For a curve through the chart origin with velocity $v$, the
  Fubini--Study speed is $\norm{v}$. Polarization gives the displayed metric. Differentiating
  the two-copy expectation and using swap symmetry gives the factor $4$.
  Projection onto $\psi^\perp$ subtracts the component $4p([\psi])\psi$.
  Multiplying $\psi$ by a phase multiplies $G$ by the same phase.
\end{lemma}
\leanproof{lem:acceptance-gradient}

\begin{lemma}[Existence and uniqueness of the global gradient flow]\label{lem:flow-existence}
  \lean{GaussianTesting.projective_gradient_flow, GaussianTesting.TensorSquare.exists_unique_smooth_projectiveFlow, GaussianTesting.TensorSquare.chartGrad, GaussianTesting.TensorSquare.IsProjectiveFlowOn, GaussianTesting.TensorSquare.IsProjectiveFlowOn.eqOn_Icc, GaussianTesting.SmoothRayCurveOn, GaussianTesting.exists_global_of_finite_extension, GaussianTesting.TensorSquare.IsProjectiveFlowOn.exists_unit_lift_Icc, GaussianTesting.TensorSquare.IsUnitFlowOn.integral_norm_velocity_sq_eq, GaussianTesting.TensorSquare.IsProjectiveFlowOn.dist_sq_le_energy, GaussianTesting.TensorSquare.IsProjectiveFlowOn.exists_limit_Ico, GaussianTesting.TensorSquare.IsProjectiveFlowOn.continue_at_finite_time}
  \leanok
  \uses{lem:acceptance-gradient}
  \noderef{r3-1}{Lemma~3.1}\par
  For every initial ray in any complex Hilbert space and every swap-commuting
  orthogonal projection on its completed tensor square,
  \paperref{flow}{Eq.~(3.3)} has a unique solution on $[0,\infty)$. Its
  coordinates are smooth in each chart through the current ray, with
  one-sided derivatives at time zero.

  \textbf{Energy and finite-endpoint continuation.}
  For the \paperref{local-flow-energy}{energy estimate in the manuscript proof},
  let $\rho$ solve \paperref{flow}{Eq.~(3.3)} in ray charts. On each closed
  subinterval $[a,b]$ it agrees with the projection of a unit-vector solution
  $\gamma$. Its velocity within that interval satisfies
  \[
    (p\circ\gamma)'(t)=\norm{\nabla p(\gamma(t))}^2,\qquad
    \int_a^b \norm{\dot\gamma(t)}^2\,dt
      = p(\gamma(b))-p(\gamma(a)).
  \]
  On a singleton interval both sides of the integral identity vanish.
  The path-length bound and Cauchy--Schwarz give
  \[
    d_{\rm FS}(\rho(a),\rho(b))^2
    \le (b-a)\bigl(p(\rho(b))-p(\rho(a))\bigr)\le b-a .
  \]
  Thus a solution on $[a,b)$, with $a<b$, converges to a ray as $t\uparrow b$.
  Restarting the local ODE at this ray extends the trajectory past $b$.
  The left and right chart derivatives agree at the joining time.

  Lean uses the Fubini--Study gradient in \bplink{lem:acceptance-gradient}{Eq.~(3.4)}.
  Multiplying a unit representative by a phase multiplies the field by
  that phase, and the field is complex-orthogonal to the representative.
  In each affine chart the velocity is the derivative of the chart map
  applied to this field at a normalized representative. A unit-vector
  solution projects to a solution of this chart ODE. Local Lipschitz
  uniqueness in the charts shows that any two ray solutions starting
  together agree on their common interval.

  Local solutions therefore agree on overlaps and define a trajectory up
  to the supremum of their end times. If that supremum were finite, the
  energy estimate and continuation argument above
  would extend the trajectory past it. This proves global existence.
  Uniqueness on closed subintervals gives uniqueness at every forward time.
\end{lemma}
\leanproof{lem:flow-existence}

\begin{lemma}[Contraction and critical-point characterization]\label{lem:contraction-critical}
  \lean{GaussianTesting.TensorSquare.contractionOperator, GaussianTesting.TensorSquare.contractionOperator_apply, GaussianTesting.TensorSquare.contractionOperator_self, GaussianTesting.TensorSquare.contractionOperator_isPositive, GaussianTesting.TensorSquare.contractionOperator_isSelfAdjoint, GaussianTesting.TensorSquare.norm_contractionOperator_le_one, GaussianTesting.TensorSquare.contractionOperator_first_moment, GaussianTesting.TensorSquare.contractionOperator_second_moment, GaussianTesting.TensorSquare.gradAccA_eq_contractionOperator, GaussianTesting.TensorSquare.gradAccA_eq_zero_iff_contractionOperator, GaussianTesting.TensorSquare.fderiv_rayAcceptance_eq_zero_iff_contractionOperator, GaussianTesting.contraction_criticalPoint_iff}
  \leanok
  \uses{def:acceptance, lem:acceptance-gradient}
  \noderef{contraction}{After Lemma~3.1}\par
  For a unit representative $\psi$ in any complex Hilbert space, define
  $J_\psi x=x\otimes\psi$. The manuscript's operator is the bounded compression
  \[
    K_\psi=J_\psi^\dagger P J_\psi
      =(I\otimes\langle\psi|)P(I\otimes|\psi\rangle).
  \]
  It is positive and self-adjoint, with $\norm{K_\psi}\le1$. Its first two
  moments are exactly
  \[
    \langle\psi,K_\psi\psi\rangle=p(\psi),\qquad
    \langle\psi,K_\psi^2\psi\rangle=q(\psi).
  \]
  The gradient is $4(K_\psi\psi-p(\psi)\psi)$, and the following are equivalent:
  the differential of projective acceptance at $[\psi]$ vanishes;
  $\nabla p(\psi)=0$; and $K_\psi\psi=p(\psi)\psi$.

  The adjoint identity identifies $J_\psi^\dagger$ with the existing Riesz
  contraction. Compressing the orthogonal projection proves positivity,
  while $J_\psi$ is an isometry for unit $\psi$ and $\norm{P}\le1$ gives
  contractivity. Applying $K_\psi$ to $\psi$ recovers the previously defined
  contracted vector. Self-adjointness identifies its squared norm with the
  second moment. Finally, \bplink{lem:acceptance-gradient}{Eq.~(3.4)} identifies
  the horizontal field with the differential's metric dual in the centered
  projective chart; its factor $4$ is nonzero, giving the critical-point equation.
\end{lemma}
\leanproof{lem:contraction-critical}

\begin{definition}[Three-copy average and pair operators]\label{lem:three-copy-average}
  \lean{GaussianTesting.threeCopyAverage, GaussianTesting.threeCopyPair12, GaussianTesting.threeCopyPair13, GaussianTesting.threeCopyPair23, GaussianTesting.threeCopyAverage_eq, GaussianTesting.threeCopyAverage_positive_contraction}
  \leanok
  \uses{def:acceptance}
  \noderef{three-copy}{Eq.~(3.6)}\par
  A two-copy acceptance projection $P$ acts on each pair of factors of the
  completed Hilbert tensor cube. For instance, the matrix element of $P_{12}$
  between $a\otimes b\otimes c$ and $x\otimes y\otimes z$ is
  $\langle a\otimes b,P(x\otimes y)\rangle\langle c,z\rangle$.
  On the completed products, $P_{12}=P\otimes I$ under the associator,
  $P_{23}=I\otimes P$, and $P_{13}$ is obtained by permuting the factors.
  The matrix elements identify these operators with their expressions in
  any Hilbert basis coordinates, using the tensor-product preliminaries.
  Thus the positive contraction
  $S=(P_{12}+P_{13}+P_{23})/3$ has the coordinate representation used in
  the moment calculation for every choice of basis.
\end{definition}

\begin{lemma}[Three-copy moment identities]\label{lem:three-copy-moments}
  \lean{GaussianTesting.threeCopyAverage_moments, GaussianTesting.hilbert_first_moment, GaussianTesting.hilbert_second_moment, GaussianTesting.hilbert_contr_coords}
  \leanok
  \uses{lem:three-copy-average}
  \noderef{r3-2}{Lemma~3.2}\par
  For a unit vector $\psi$ in any complex Hilbert space put
  $q(\psi)=\norm{(I\otimes\langle\psi|)P\psi^{\otimes2}}^2$. Then
  \[
    \langle\psi^{\otimes3},S\psi^{\otimes3}\rangle=p(\psi),
    \qquad \langle\psi^{\otimes3},S^2\psi^{\otimes3}\rangle=\frac{p(\psi)+2q(\psi)}{3}.
  \]
  Both proofs expand the three-copy average. The diagonal projection terms give
  $p$, and the six cross terms give $q$ by permutation symmetry. Lean performs
  this expansion in Hilbert basis coordinates and uses the Riesz identity
  for the contraction to transfer $q$ back to the original space.
\end{lemma}
\leanproof{lem:three-copy-moments}

\begin{lemma}[Fixed tensor cubes and certainty states]\label{lem:triple-certainty}
  \lean{GaussianTesting.threeCopyAverage_fixed_iff_accProb_eq_one, GaussianTesting.threeCopyAverage_spectrum_subset_unitInterval, GaussianTesting.threeCopyAverage_hasEigenvalue_one, GaussianTesting.threeCopyAverage_isGreatest_eigenvalue}
  \leanok
  \uses{def:fidelity, lem:three-copy-average, lem:three-copy-moments}
  \noderef{triple-certainty}{Section~3, fixed tensor cubes}\par
  The three-copy average $S$ has spectrum in $[0,1]$. For any unit vector
  $\psi$, let $v=\psi^{\otimes3}$. Then $Sv=v$ if and only if $p(\psi)=1$.

  Conversely, if $Sv=v$, the first moment is $\norm{v}^2=1$.
  Thus any certainty state supplies a nonzero eigenvector at $1$.
  When the certainty set is nonempty, $1$ is the largest eigenvalue.
\end{lemma}

\begin{lemma}[Moment and gradient bounds from the spectrum]\label{lem:bound-gradient}
  \lean{GaussianTesting.lojasiewiczData_of_threeCopy_spectrum, GaussianTesting.norm_gradAcc_sq_ge_of_threeCopy_spectrum, GaussianTesting.symmetricCube, GaussianTesting.threeCopyAverage_maps_symmetricCube, GaussianTesting.symmetricThreeCopyAverage, GaussianTesting.symmetricThreeCopyAverage_positive_contraction, GaussianTesting.symmetricThreeCopyAverage_moments, GaussianTesting.lojasiewiczData_of_symmetricThreeCopy_spectrum, GaussianTesting.norm_gradAcc_sq_ge_of_symmetricThreeCopy_spectrum, GaussianTesting.spectrum_gap_iff_polynomial_nonneg, GaussianTesting.spectrum_gap_iff_quadraticForm_nonneg, GaussianTesting.SpectralGapHyp}
  \leanok
  \uses{lem:three-copy-moments, lem:gradient-identity}
  \noderef{r3-3}{Lemma~3.3}\par
  Let $P$ be an orthogonal two-copy projection commuting with swap and
  $S=(P_{12}+P_{13}+P_{23})/3$ on the completed tensor cube of a complex
  Hilbert space. If $\mathrm{spec}(S)\subseteq[0,\alpha]\cup\{1\}$ for
  $0\le\alpha\le1$, then
  $\norm{\nabla p}^2\ge16(p-\tfrac32\alpha)(1-p)$ on unit vectors.

  \textbf{Spectral equivalence.}
  The \paperref{spectral-gap}{spectral hypothesis} can be expressed as a
  polynomial inequality. Let $S$ be a positive contraction on a complex Hilbert space and $\alpha\in\R$.
  Then $\mathrm{spec}_{\C}(S)\subseteq [0,\alpha]\cup\{1\}$ if and only if
  $(S-I)(S-\alpha I)\ge0$, equivalently if and only if
  $\operatorname{Re}\langle v,(S-I)(S-\alpha I)v\rangle\ge0$ for every $v$.
  By continuous functional calculus the polynomial is positive precisely when
  $(t-1)(t-\alpha)\ge0$ on the spectrum of $S$, which lies in $[0,1]$.
  The finite predicate \texttt{SpectralGapHyp} uses this quadratic form for the
  \paperref{three-copy}{three-copy average, Eq.~(3.6)}.

  The \paperref{spectral-gap}{spectral hypothesis} implies
  polynomial positivity for the three-copy average. Lean evaluates
  $(S-I)(S-\alpha I)\ge0$ on $\psi^{\otimes3}$ and uses
  \paperref{r3-2}{Lemma~3.2} to get $q\ge p-\tfrac32\alpha(1-p)$.
  The \paperref{gradient-norm}{gradient identity, Eq.~(3.5)} gives the
  claimed gradient bound.
  This is the moment inequality used by \paperref{r3-4}{Theorem~3.4}.

  \textbf{Symmetric-cube strengthening.}
  The \paperref{symmetric-cube-gap}{footnote immediately before Lemma~3.3}
  is also checked: it suffices to assume the spectral inclusion for
  $S|_{\operatorname{Sym}^3(\mathcal H)}$.
  Lean constructs this closed subspace as the common fixed space of the
  adjacent factor swaps, proves that $S$ commutes with both swaps, and
  restricts $S$ to it. The restriction is a positive contraction and has the
  same two moments on $\psi^{\otimes3}$. Applying the preceding polynomial
  argument on this subspace proves the same moment and gradient bounds.
\end{lemma}
\leanproof{lem:bound-gradient}

  \leanok
  \uses{lem:flow-existence, lem:bound-gradient, prelim:flow}
  \noderef{r3-4}{Theorem~3.4}\par
  \textbf{Proof status.} Both conclusions are proved from the projective
  gradient flow of Lemma~3.1, its path-length estimate, and its certainty
  limit, following the manuscript's argument.

  Let $P$ be an orthogonal projection on $\mathcal H^{\otimes2}$ commuting
  with swap, with nonempty \paperref{fidelity}{certainty set (Eq.~(3.2))} $\mathcal G$.
  Form $S=(P_{12}+P_{13}+P_{23})/3$ on $\mathcal H^{\otimes3}$ and assume
  $\mathrm{spec}(S)\subseteq[0,\alpha]\cup\{1\}$ for $0\le\alpha<2/3$.
  Put $C_\alpha=(1-3\alpha/2)^{-1}$.

  For each unit input $\psi$, there is one $\phi\in\mathcal G$ with both
  properties. When $p(\psi)>3\alpha/2$,
  \[
    \arccos F_{\mathcal G}(\psi)
    \le \arccos|\langle\psi,\phi\rangle|
    \le \tfrac12\arcsin\sqrt{C_\alpha(1-p(\psi))}.
  \]
  For every $\tau\ge0$ such that $p(\psi)\ge1-\tau$,
  \[
    1-C_\alpha\tau\le\fid{\psi}{\phi}\le F_{\mathcal G}(\psi)^2.
  \]

  \textbf{\paperref{energy-identity}{Eq.~(3.7)}: acceptance derivative and speed.}
  Let $\rho_t$ solve the ray gradient flow on $[0,\infty)$, and put
  $p_t=p(\rho_t)$. If $\mathrm{spec}(S)\subseteq[0,\alpha]\cup\{1\}$
  with $0\le\alpha\le1$, then for every $t\ge0$,
  \[
    \begin{aligned}
      p'_t &= \norm{\dot\rho_t}_{\rm FS}^{2}=\norm{\nabla p(\rho_t)}^2 \\
            &\ge 16(p_t-\tfrac32\alpha)(1-p_t)
             \ge 16(p_0-\tfrac32\alpha)(1-p_t).
    \end{aligned}
  \]
  The metric speed is the limit of $d_{\rm FS}(\rho_r,\rho_t)/|r-t|$
  as $r\to t$ within the forward time domain. At $t=0$ both the speed and
  the derivative are taken from the right. A horizontal unit-vector lift
  gives the speed and the acceptance derivative. The first inequality is
  the pointwise gradient bound, and the second uses $p_0\le p_t\le1$.

  \textbf{Projective flow and acceptance.}
  Start the global projective solution of
  \bplink{lem:flow-existence}{Lemma~3.1} at the input ray.
  For $3\alpha/2<p_0<1$, the
  \paperref{flow-acceptance-limit}{strict increase and convergence claim}
  gives, for every $t\ge0$,
  \[
    1-p_t\ge e^{-16t}(1-p_0)>0,
    \qquad p_t<1,
    \qquad \lim_{t\to\infty}p_t=1,
  \]
  and $p_s<p_t$ whenever $0\le s<t$. The upper derivative bound
  $p'_t\le16(1-p_t)$ gives the exponential estimate.
  \paperref{energy-identity}{Eq.~(3.7)} then makes $p'_t$ positive.
  Monotonicity and boundedness give a limit. A limit below $1$ would give
  a uniform positive lower bound on $p'_t$, contradicting boundedness.
  The limit is proved for real forward time.

  \textbf{Path length and certainty limit.}
  The \paperref{flow-ray-limit}{trajectory convergence claim}
  holds for every such forward projective solution. Put
  $c=3\alpha/2$ and $V(p)=\tfrac12\arcsin\sqrt{(1-p)/(1-c)}$.
  For $0\le s\le t$, Lean proves
  \[
    d_{\mathrm{FS}}(\psi_s,\psi_t)
    \le \int_s^t \norm{\nabla p(\psi_r)}\,\mathrm{d}r
    \le V(p_s)-V(p_t)\le V(p_s).
  \]
  A horizontal lift and the partition estimate for angles give the first
  inequality. The chain rule and the fundamental theorem of calculus for
  $V(p_t)$ express the manuscript's change of variables and evaluation of
  the integral; Eq.~(3.7) gives the second inequality.
  Since $p_s\to1$ and $V(1)=0$, the trajectory is Cauchy as real time tends
  to infinity. Completeness gives a limiting ray $\psi_\infty$.
  Continuity of acceptance gives $p(\psi_\infty)=1$, and continuity of
  distance gives $d_{\mathrm{FS}}(\psi_s,\psi_\infty)\le V(p_s)$.

  \textbf{Angular and linear fidelity bounds.}
  Choose a normalized representative $\phi$ of the limiting certainty ray.
  Its distance from the initial ray equals
  $\arccos|\langle\psi,\phi\rangle|$, giving the angular witness bound.
  The supremum defining fidelity is at least this witness's absolute
  overlap, so its arccosine is at most the witness angle.
  The manuscript estimate
  \[
    \cos^2\!\left(\tfrac12\arcsin\sqrt{x}\right)
    =\tfrac12\left(1+\sqrt{1-x}\right)\ge1-x
    \quad(0\le x\le1)
  \]
  gives the linear bound for the same $\phi$, independently of $\tau$.
  The squared witness overlap is at most squared fidelity.
  If $p(\psi)=1$, take $\phi=\psi$. If $p(\psi)\le3\alpha/2$, the
  angular hypothesis is false and the linear lower bound is nonpositive,
  so any member of the standing nonempty certainty set suffices.
\end{theorem}

  of the three projections lies in all three axis-fixed spaces: the norm
  inequalities force each projected vector to have the original norm.
  Coordinate-plane rotations generate $\mathrm{SO}(3)$, so this is precisely
  the invariant space of the full rotation action.

  any unit vector to the first coordinate vector. Haar averaging a continuous
  observable therefore gives a constant function, by left invariance of
  Haar measure. Vector-valued Haar averaging is a bounded operator on $L^2$.
  Density of continuous observables and closedness of the span of $1$ show
  that its image lies in that span. An invariant vector equals its own
  average, and constants are invariant. Thus

    \ker(K-I)=\C\,1,

  \]

  contained in $[0,7/12]$, and positivity gives operator norm at most $7/12$.

  orthogonal complement.

  has full support, so it is a nonzero $L^2$ vector. Symmetry and its
  eigenvalue different from $1$ place it in $(\C\,1)^\perp$.
  Thus the restricted operator has norm and largest spectral value exactly
  $7/12$, giving the spectral gap $1-7/12=5/12$.
\end{theorem}

\begin{theorem}[Gaussian-square span and optimality]\label{target:span-optimality}
  \lean{GaussianTesting.fermionicGaussianSquareSpan_eq_kernel, GaussianTesting.swap2_PF, GaussianTesting.closedSpan_projection_le_effect, GaussianTesting.fermionicTest_projection_le_effect, GaussianTesting.fermionicTest_acceptance_le, GaussianTesting.meanZeroGaussianSquareSpan, GaussianTesting.meanZeroGaussianSquareSpan_eq_symmetric_kernel, GaussianTesting.mem_meanZeroGaussianSquareSpan_iff, GaussianTesting.polynomial_mem_copyGramAlgebra_of_invariant, GaussianTesting.polynomialGaussian_mem_meanZeroGaussianSquareSpan, GaussianTesting.meanZeroGaussianSquareSpan_projection_square, GaussianTesting.bosonicMeanZeroGaussian_optimality}

  \uses{lem:exact-fermions, target:fig1-fermions, prp:mean-zero-gaussian, prelim:bosons}
  \noderef{span-optimality}{Section~1, span and optimality assertion}\par
  The Gaussian tensor squares span the accepting subspace, with the symmetric
  subspace restriction in the bosonic case. Their orthogonal projection is
  the least positive contraction with perfect completeness, so its acceptance
  is minimal on every identical two-copy input.

  \textbf{Fermionic span.}
  The \paperref{fermion-kernel-span}{fermionic kernel} is exactly the linear
  span of Gaussian squares from both parity classes, for every mode count.
  Consequently $P_F$ is supported on the symmetric subspace. The span is
  invariant under $c_j\otimes c_j$ and simultaneous Gaussian rotations.
  The joint Bell eigenlines are one-dimensional. A nonzero overlap with the
  vacuum square supplies one balanced eigenline; signed coordinate
  permutations supply all balanced eigenlines, which span $\ker\Lambda_F$.
  This proves the span statement directly. The manuscript specifies twirling
  over the full orthogonal group in the cited argument, including both parity
  classes. The checked proof uses the previously agreed invariant-span argument.

  For any Hilbert space, a positive contraction with certainty on a set
  fixes its closed span. Its compression to the orthogonal complement is
  positive, so it dominates the span projection. Applied to the checked
  fermionic span, this proves optimality even on arbitrary two-copy vectors.

  \textbf{Bosonic span.} On actual $L^2(\mathbb R^{2m})$, the closed span of
  centered Gaussian squares is exactly
  $\ker\Lambda\cap\{f:Sf=f\}$, where $S$ interchanges the two copies.
  This holds for every mode count, including zero, without a moment
  assumption on the kernel vectors. Rotation averaging preserves polynomial
  Gaussians because each polynomial orbit lies in a finite-dimensional span.
  Averaging also over swap produces polynomials invariant under both actions.
  In complex coordinates $u=x+iy$, $v=x-iy$, these are balanced polynomials
  fixed by interchange of $u,v$. Finite-dimensional duality and the joint
  kernel of the symmetric mixed derivatives show that they are generated by
  $u_i v_j+u_j v_i$, hence by $x_i x_j+y_i y_j$ in real coordinates.

  Identical quadratic phase operators preserve the Gaussian-square span.
  Differentiating their matrix coefficients, with an integrable dominating
  function, puts each Gram quadratic times a polynomial Gaussian in that
  span. Algebra induction starts from the vacuum square. Density of
  polynomial Gaussians and continuity of rotation/swap averaging then give
  every symmetric kernel vector. The reverse inclusion uses
  Proposition~6.1 and symmetry of identical squares.

  On identical inputs, the actual $P_0$ agrees with this span projection:
  $P_0$ commutes with swap, so its output remains symmetric. The general
  positive-contraction argument therefore proves the manuscript's bosonic
  optimality assertion. The alternative effect needs no swap invariance.
  The manuscript's wording ``measures exactly the span'' is retained in
  its stated identical two-copy-input context.
\end{theorem}

\begin{theorem}[Pure-state trace distance]\label{lem:pure-state-trace-distance}
  \lean{GaussianTesting.pureStateDifference, GaussianTesting.traceNorm, GaussianTesting.pureStateTraceDistance, GaussianTesting.pureStateDifference_cube, GaussianTesting.abs_pureStateDifference, GaussianTesting.isPositiveTraceClass_abs_pureStateDifference, GaussianTesting.traceNorm_pureStateDifference, GaussianTesting.pureStateTraceDistance_eq_sqrt, GaussianTesting.pureStateTraceDistance_eq_half_traceNorm, GaussianTesting.le_pureStateTraceDistance_iff}
  \noderef{trace-distance}{After Theorem~1.1}\par
  \leanok
  For unit vectors in any complex Hilbert space,
  \[
    D(\psi,\phi)=\frac12\bigl\||\psi\rangle\langle\psi|
      -|\phi\rangle\langle\phi|\bigr\|_1
      =\sqrt{1-|\langle\psi,\phi\rangle|^2}.
  \]
  The trace norm is defined as $\operatorname{Tr}|A|$, with
  $|A|=\sqrt{A^\dagger A}$. For the difference $A$ of the two rank-one
  projections, Lean proves that $|A|$ is positive trace class and that its
  trace is independent of the Hilbert basis. No finite-dimensional
  assumption is made.

  Put $d=\sqrt{1-|\langle\psi,\phi\rangle|^2}$.
  The rank-one calculation gives $A^3=d^2A$ and
  $\operatorname{Tr}A^2=2d^2$. For $d>0$, positivity and uniqueness of
  the square root give $|A|=d^{-1}A^2$, hence $\operatorname{Tr}|A|=2d$.
  For $d=0$, the rank-one projections coincide.
  Thus, for every $\varepsilon\ge0$, $D(\psi,\phi)\ge\varepsilon$
  is equivalent to $|\langle\psi,\phi\rangle|^2\le1-\varepsilon^2$.
\end{theorem}
\leanproof{lem:pure-state-trace-distance}

\begin{theorem}[Testing Gaussianity with constant sample complexity]\label{target:measurement-endpoints}
  \lean{GaussianTesting.fermionic_gaussian_testing, GaussianTesting.bosonic_gaussian_testing, GaussianTesting.fermionic_gaussian_testing_traceDistance, GaussianTesting.bosonic_gaussian_testing_traceDistance, GaussianTesting.pureFidelity_sq_le_iff, GaussianTesting.fermionicTestCopies_le_rate, GaussianTesting.testingCopies_le_rate}
  \uses{cor:fermion-testing, cor:boson-testing, lem:pure-state-trace-distance, def:fidelity, prp:general-certainty, prelim:fermions, prelim:bosons, target:fig1-fermions, target:fig1-mean-zero, target:fig1-coherent, target:fig1-general}
  \noderef{r1-1}{Theorem~1.1}\par
  \leanok
  For $\varepsilon,\delta\in(0,1)$, bosonic and fermionic systems admit tests
  using $O(\varepsilon^{-2}\log(1/\delta))$ copies of a pure state $\psi$.
  Each test always accepts a Gaussian state. If every Gaussian state
  $\phi$ has $\fid{\psi}{\phi}\le1-\varepsilon^2$, the test rejects with
  probability at least $1-\delta$.

  The physical bosonic classes and all three bosonic
  circuits, followed by independent repetition with exact copy counts,
  are established in \paperref{r6-8}{Corollary~6.8}.
  The independent fermionic Gaussian class, the manuscript proofs of
  Lemma~5.1 and Theorem~5.2, and the repeated pair-test protocol with exact
  copy counts are established in \paperref{r5-3}{Corollary~5.3}.
  The accepting Bell-readout projection in \paperref{f1a}{Figure~1(a)}
  is identified with $P_F$.

  The pointwise overlap promise is equivalent to the bound on squared
  class fidelity in the corollaries. The
  \paperref{trace-distance}{rank-one trace-distance identity}
  gives exactly the interpretation as distance at least $\varepsilon$
  from every Gaussian state, with the same protocols.

  For fermions take $(k,C)=(2,8)$, and for general bosons take
  $(k,C)=(4,48)$. The exact copy count is
  $k\lceil(C/\varepsilon^2)\log(1/\delta)\rceil$, bounded uniformly by
  $kC\varepsilon^{-2}\log(1/\delta)+k$.
  For $\delta\le e^{-1}$ it is at most
  $2kC\varepsilon^{-2}\log(1/\delta)$, independently of the number of modes.
  The additive round in the uniform bound is the ceiling qualification
  recorded in the manuscript's source comment. The bosonic interface also
  includes mean-zero Gaussian and coherent states, with $(k,C)=(2,8),(2,4)$.
\end{theorem}
\leanproof{target:measurement-endpoints}
